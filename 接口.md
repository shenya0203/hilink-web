工业路由器的配置参数说明
1. 串口配置
配置文件位置：/etc/config/uart
配置文件参数说明：
config uart 'Uart1'
    option enable '1'       # 使能  页面没有对应配置 默认开启
    option name 'Uart1'     # 名称  对应第一个配置
    option work_mode '2'    # 工作模式  页面没有对应参数 默认2
    option baud_rate '57600' # 波特率  默认57600
    option data_bit '8'     # 数据位  默认8
    option stop_bit '1'     # 停止位  默认1
    option parity '0'       # 校验位  默认0
    option pack_len '1460'  # 帧打包长度 默认1460
    option pack_time '0'    # 帧打包时间 默认0
    option func '1'         # 功能  页面无对应配置
    option select '0'       # 选择  页面无对应配置
    option device '/dev/ttyS0'  # 设备  默认/dev/ttyS0 对应配置的第一个串口

config uart 'Uart2'
